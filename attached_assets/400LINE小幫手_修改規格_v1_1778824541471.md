### 1.6 修改 #4: tri-element 門檻 12 → 16 + 訊號要求

**檔案**: `server/services/announcement/announcementRuleEngine.ts`
**位置**: 第 178 行

**改寫**:

```ts
// ── 三元素至少命中兩項且分數夠高 → rule_matched ───────────────────────
const triCount = [score.hasTimeSensitivity, score.hasExplicitAudience, score.hasExplicitAction]
  .filter(Boolean).length;

// 需要 triCount >= 2 + total >= 16 + 至少一個結構/scope/強訊號
const hasTriSignal = score.hasStructure
  || score.scopeType !== 'group'
  || strongHits.length > 0;

if (triCount >= 2 && score.total >= 16 && hasTriSignal) {
  matched.push('tri_element_match');
  if (score.hasTimeSensitivity) matched.push('has_time');
  if (score.hasExplicitAudience) matched.push('has_audience');
  if (score.hasExplicitAction) matched.push('has_action');
  return { decision: 'rule_matched', reason: 'tri_element', score, matchedRules: matched };
}
```

**驗收**: 「明天各館同仁請確認本日值班」(時間+對象+對象+對象+行動，score=14, no structure, scope=multi_facility) → 16 分仍不到，但 scope!=group 命中 hasTriSignal，需重新測試是否還是會過。預期不該過。

### 1.8 修改 #6: 開頭/結尾噪音詞加重扣分

**檔案**: `server/services/announcement/announcementRuleEngine.ts`
**位置**: `scoreMessage()` 第 128 行

**改寫**:

```ts
// 噪音詞扣分：開頭或結尾命中 ×2 權重
const HEAD_TAIL_LEN = 6;
const head = text.substring(0, HEAD_TAIL_LEN);
const tail = text.substring(Math.max(0, text.length - HEAD_TAIL_LEN));
let noiseScore = 0;
for (const noise of NOISE_WORDS) {
  if (head.includes(noise) || tail.includes(noise)) {
    noiseScore += 8;  // 原本 -4，邊緣 ×2
  } else if (text.includes(noise)) {
    noiseScore += 4;
  }
}
total -= noiseScore;
```

**驗收**: 「謝謝主管即日起會配合」(10 即日起 - 8 開頭謝謝 = 2 分) 不再到 score threshold 6。

## 2. 服務監聽 + 白名單外部化

### 2.1 整體架構

```
┌───────────────────────────────┐         ┌─────────────────────────┐
│   400LINE 小幫手 (Replit)      │         │  400 儀表板 IT 端         │
│                                │         │  (另一個 repo)            │
│  ┌──────────────────────────┐ │         │                         │
│  │ Webhook Pusher           │─┼─POST───►│ /webhook/lineXBS        │
│  │  - metric_heartbeat 60s  │ │         │ (HMAC 驗證)              │
│  │  - error_event 即時       │ │         │                         │
│  │  - decision_event (opt)  │ │         │                         │
│  └──────────────────────────┘ │         │                         │
│                                │         │                         │
│  ┌──────────────────────────┐ │         │                         │
│  │ Admin REST API           │◄┼─GET─────│ on-demand 查詢            │
│  │  - whitelist CRUD        │ │  POST   │  (錯誤 log/最近判決)      │
│  │  - decision trace tail   │ │  PATCH  │                         │
│  │  - error log query       │ │  DELETE │                         │
│  └──────────────────────────┘ │         │                         │
└───────────────────────────────┘         └─────────────────────────┘
```

**設計原則**:
- **可量化指標** (pipeline stats, Gemini health, webhook health, cron status) → 用 push（每分鐘 heartbeat）
- **離散事件** (錯誤、Gemini 逾時、簽章失敗) → 用 push（即時）
- **互動式查詢** (錯誤 log 列表、訊息流 tail、白名單管理) → 用 pull（REST API）

### 2.4 白名單存取層（取代 announcementConfig.ts 的硬編碼）

新增 `server/services/admin/whitelistRepo.ts`：

```ts
import { db } from '../../db';
import { adminWhitelist, vipWhitelist, groupTierConfig } from '@shared/schema';
import { eq, and } from 'drizzle-orm';

// In-memory cache（5 分鐘 TTL）
let cache: {
  admins: Map<string, AdminRecord>;
  vips: Map<string, VipRecord>;
  groups: Map<string, GroupTierRecord>;
  loadedAt: number;
} | null = null;

const CACHE_TTL_MS = 5 * 60 * 1000;

async function loadCache() {
  const [admins, vips, groups] = await Promise.all([
    db.select().from(adminWhitelist).where(eq(adminWhitelist.enabled, true)),
    db.select().from(vipWhitelist).where(eq(vipWhitelist.enabled, true)),
    db.select().from(groupTierConfig).where(eq(groupTierConfig.enabled, true)),
  ]);
  cache = {
    admins: new Map(admins.map(r => [r.userId, r])),
    vips: new Map(vips.map(r => [r.userId, r])),
    groups: new Map(groups.map(r => [r.groupId, r])),
    loadedAt: Date.now(),
  };
}

export async function getCache() {
  if (!cache || Date.now() - cache.loadedAt > CACHE_TTL_MS) {
    await loadCache();
  }
  return cache!;
}

export function invalidateCache() { cache = null; }

// 對外查詢介面
export async function isAdminWhitelisted(userId: string): Promise<boolean> {
  const c = await getCache();
  return c.admins.has(userId);
}

export async function getVipName(userId: string): Promise<string | null> {
  const c = await getCache();
  return c.vips.get(userId)?.displayName ?? null;
}

export async function getGroupTier(groupId: string): Promise<'A' | 'B' | 'C' | null> {
  const c = await getCache();
  return c.groups.get(groupId)?.tier as any ?? null;
}

export async function getFacilityName(groupId: string): Promise<string | null> {
  const c = await getCache();
  return c.groups.get(groupId)?.facilityName ?? null;
}

// CRUD（給 Admin API 用）
export async function listAdmins() { /* ... */ }
export async function upsertAdmin(record: AdminRecord) { /* ... */ invalidateCache(); }
export async function deleteAdmin(userId: string) { /* ... */ invalidateCache(); }
// vip / group 同上
```

### 2.6 Webhook Push 機制

新增 `server/services/monitoring/dashboardPusher.ts`：

```ts
import crypto from 'crypto';

interface PushPayload {
  event_type: 'metric_heartbeat' | 'error_event' | 'decision_event';
  timestamp: string;
  source: 'lineXBS';
  payload: unknown;
}

const URL = process.env.DASHBOARD_WEBHOOK_URL;
const SECRET = process.env.DASHBOARD_WEBHOOK_SECRET ?? '';
const ENABLED = process.env.DASHBOARD_WEBHOOK_ENABLED !== 'false';

// 簡易 circuit breaker —— 連續失敗 5 次後暫停 5 分鐘
let consecutiveFailures = 0;
let pausedUntil = 0;

export async function pushToDashboard(payload: PushPayload): Promise<void> {
  if (!ENABLED || !URL) return;
  if (Date.now() < pausedUntil) return;

  const body = JSON.stringify(payload);
  const signature = crypto.createHmac('sha256', SECRET).update(body).digest('hex');

  try {
    const resp = await fetch(URL, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'X-Signature': `sha256=${signature}`,
        'X-Event-Type': payload.event_type,
      },
      body,
      signal: AbortSignal.timeout(5000),
    });

    if (!resp.ok) throw new Error(`HTTP ${resp.status}`);
    consecutiveFailures = 0;
  } catch (err) {
    consecutiveFailures++;
    if (consecutiveFailures >= 5) {
      pausedUntil = Date.now() + 5 * 60 * 1000;
      console.error('🔴 [dashboard] circuit breaker tripped, paused 5 min');
    }
  }
}

// 心跳排程
export function startHeartbeat(): void {
  const interval = parseInt(process.env.DASHBOARD_HEARTBEAT_INTERVAL_MS ?? '60000');
  setInterval(async () => {
    const { getCurrentSnapshot } = await import('./healthAggregator');
    const snapshot = await getCurrentSnapshot();
    await pushToDashboard({
      event_type: 'metric_heartbeat',
      timestamp: new Date().toISOString(),
      source: 'lineXBS',
      payload: snapshot,
    });
  }, interval);
}
```
